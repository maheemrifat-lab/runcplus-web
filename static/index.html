POST /api/run 
curl -X POST https://<current-host>/api/run \
  -H "Content-Type: application/json" \
  -d '{
    "code": "#include <iostream>\nint main() { std::cout << \"Hello RunCplus!\" << std::endl; return 0; }",
    "stdin": "",
    "standard": "c++20",
    "optimization": "-O2"
  }'
